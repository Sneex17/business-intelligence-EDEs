/*1-) Top 25 sucursales con mayor perdida de energía por año*/
select {[Measures].[Energia Entregada G Wh], 
[Measures].[Energia Perdida G Wh]} on columns,
non empty {[Dim EDE].[Nombre].[Nombre] * topcount([Dim Sucursal].[Nombre].[Nombre],25,[Measures].[Energia Perdida G Wh])} on rows
from [EDEs_EnergiaCube]
where [Dim Tiempo].[Anio].&[2020];


/*2-) Top 5 provincias con mayor volumen de energía 
entregada en el último año registrado*/
select {[Measures].[Energia Entregada G Wh]} on columns,
non empty {[Dim Ubicacion].[Region].[Region] * topcount({[Dim Ubicacion].[Provincia].[Provincia]},
5,[Measures].[Energia Entregada G Wh])} on rows
from [EDEs_EnergiaCube]
where [Dim Tiempo].[Anio].&[2026];


/*3-) Top 1, de EDE que entrego mayor energía durante el 
primer trimestre (Trimestre 1) de cualquier año.*/
select non empty {[Dim Tiempo].[Anio].[Anio] * 
topcount([Dim EDE].[Nombre].[Nombre],1,[Measures].[Energia Entregada G Wh])} on columns,
non empty {[Dim Tiempo].[Id Trimestre].&[1]} on rows
from [EDEs_EnergiaCube];


/*4-) Obtener la lista de municipios donde el PorcentajeDePerdida haya sido estrictamente 
superior al 20% (0.20) durante el año 2020, mostrando en las columnas la Energia Entregada GWh 
y la Energia Perdida GWh*/
select {[Measures].[Energia Entregada G Wh],[Measures].[Energia Perdida G Wh],
[Measures].[PorcentajeDePerdida]} on columns,
non empty filter([Dim Ubicacion].[JerarquiaMunicipio].[Municipio].members,
[Measures].[PorcentajeDePerdida] > 0.20) on rows
from [EDEs_EnergiaCube]
where ([Dim Tiempo].[Anio].&[2020]);


/*5-) Crear un miembro calculado llamado [Energia Perdida Ano Anterior] usando la función 
PARALLELPERIOD para comparar la Energia Perdida GWh del mes de Mayo de 2023 con la pérdida 
registrada en el mismo mes (Mayo) pero del año anterior (2022), mostrando ambas métricas en 
las columnas y las EDEs en las filas.*/
with 
member [Measures].[Energia Perdida Ano Anterior] as
(parallelperiod([Dim Tiempo].[JerarquiaYear].[Anio], 1,
[Dim Tiempo].[JerarquiaYear].CurrentMember),
[Measures].[Energia Perdida G Wh])

select {[Measures].[Energia Perdida G Wh], 
[Measures].[Energia Perdida Ano Anterior],[Measures].[PorcentajeDePerdida]} on columns,
non empty [Dim EDE].[Nombre].[Nombre].members on rows
from [EDEs_EnergiaCube]
where ([Dim Tiempo].[JerarquiaYear].[Anio].&[2023],[Dim Tiempo].[Id Mes].&[5]);


/*6-) Obtener el volumen total de energía entregada (en GWh) y el porcentaje de participación/contribución 
de cada EDE con respecto al total general del país durante el año 2024, formateando el resultado de la 
participación como un porcentaje.*/
with 
member [Measures].[Porcentaje Participacion EDE] as
  ([Dim EDE].[Nombre].CurrentMember, [Measures].[Energia Entregada G Wh]) / 
  ([Dim EDE].[Nombre].[All], [Measures].[Energia Entregada G Wh]),
  format_string = "Percent"

select
  {[Measures].[Energia Entregada G Wh], 
   [Measures].[Porcentaje Participacion EDE]} on columns,
  non empty [Dim EDE].[Nombre].[Nombre].members on rows
from [EDEs_EnergiaCube]
where ([Dim Tiempo].[Anio].&[2024]);


/*7-) Ranking de Empresas Distribuidoras (EDEs) según el Porcentaje de Entrega de Energía (Año 2024)*/
with 
set [RankingEDEs] as
  order([Dim EDE].[Nombre].&[EDEESTE] : [Dim EDE].[Nombre].&[EDESUR], [Measures].[PorcentajeDeEntrega], bdesc)

member [Measures].[Posicion Rank] as
  rank([Dim EDE].[Nombre].CurrentMember, [RankingEDEs])

select 
  {[Measures].[Posicion Rank], 
   [Measures].[PorcentajeDeEntrega]} on columns,
  [RankingEDEs] on rows
from [EDEs_EnergiaCube]
where ([Dim Tiempo].[Anio].&[2024]);

/*8-) Sucursales Críticas con Pérdida de Energía Superior a 20 GWh (Año 2023)*/
select 
  {[Measures].[Energia Perdida G Wh], 
   [Measures].[PorcentajeDePerdida]} on columns,
  non empty filter(
    [Dim Sucursal].[Nombre].[Nombre].members,
    [Measures].[Energia Perdida G Wh] > 20.0
  ) on rows
from [EDEs_EnergiaCube]
where ([Dim Tiempo].[Anio].&[2023]);

/*9-) Dinero generado en cada mes del año 2021 por ede*/
select {[Tiempo].[Id Mes].&[1] : [Tiempo].[Id Mes].&[12]} on columns,
non empty [EDE].[Nombre].[Nombre] * [Measures].[Monto Total] on rows
from [FacturasSucursalesCube]

/*10-) Obtener el monto total facturado por cada EDE y mes durante el año 2024, comparándolo contra el mismo período 
del año anterior mediante ParallelPeriod, incluyendo la variación monetaria y el porcentaje de crecimiento interanual.*/
with 
member [Measures].[Monto Total Anio Anterior] AS
(ParallelPeriod([Tiempo].[Anio].[Anio],1,[Tiempo].[Anio].CurrentMember),
[Measures].[Monto Total])

member [Measures].[Variacion Monto] AS
[Measures].[Monto Total] - [Measures].[Monto Total Anio Anterior]

member [Measures].[Crecimiento %] AS
iif(ISNULL([Measures].[Monto Total Anio Anterior]) or [Measures].[Monto Total Anio Anterior] = 0,
null,([Measures].[Monto Total] - [Measures].[Monto Total Anio Anterior]) / [Measures].[Monto Total Anio Anterior]),
format_string = "Percert"

select {[Measures].[Monto Total],[Measures].[Monto Total Anio Anterior],
[Measures].[Variacion Monto],[Measures].[Crecimiento %]} on columns,
non empty ([EDE].[Nombre].[Nombre].Members * [Tiempo].[Id Mes].[Id Mes].Members) on rows
from [FacturasSucursalesCube]
where ([Tiempo].[Anio].&[2024]);
